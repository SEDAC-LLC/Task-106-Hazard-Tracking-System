03

Parametric Model

SysML ConstraintBlocks · HTS analytics pipeline · completeness scoring
// ConstraintBlock: HTSCompleteness [per MIL-STD 882E §101.2.6 MDEs]
mdeScore[field] = |{h : h.field ≠ ∅}| / |H|
overallCompleteness = Σf∈MDE mdeScore[f] / |MDE|
criticalMDEScore = mean(mdeScore[f] : f ∈ {hazard_id, causal_factor, severity_initial, risk_acceptance, verification_method})

// ConstraintBlock: RACTrendAnalysis
racDistInit[r] = |{h : rac(h.sev_init, h.prob_init) = r}|
racDistResid[r] = |{h : rac(h.sev_resid, h.prob_resid) = r}|
riskReductionIndex = (Σ rank(rac_init) − Σ rank(rac_resid)) / Σ rank(rac_init)

// ConstraintBlock: ClosureVelocity
closureRate = |{h : h.status ∈ {closed, accepted}}| / |H|
openHighSerious = |{h : racRank(h.rac_initial) ≥ 3 ∧ h.status = open}|

// ConstraintBlock: AcceptanceAudit
unacceptedHigh = {h : racRank(h.rac_residual) ≥ 2 ∧ risk_acceptance = ∅}

// ConstraintBlock: ControlEffectiveness
ctlCoverage[h] = |{c ∈ C : h.hazard_id ∈ c.hazard_ids}|
unverifiedHighRisk = |{c : c.verif_status ≠ complete ∧ ∃ h ∈ c.hazards : racRank(h) ≥ 3}|
